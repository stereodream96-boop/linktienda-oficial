  Link Tienda — Documentación completa

  1) Descripción del proyecto
  ----------------------------
  Link Tienda es una aplicación ligera para mostrar la presencia online de una tienda o salón: frontend en React (Vite) y backend en PHP pensado para ejecutarse sobre XAMPP/Apache+MySQL. El sistema permite crear páginas públicas por `slug`, con hero/cover, secciones dinámicas (productos o servicios), una sección de categorías con imágenes y un panel admin (APIs) para subir imágenes y gestionar inventario/servicios.

  2) Qué se ha implementado (resumen de puntos realizados)
  -----------------------------------------------------
  - Página pública con `PageViewer.jsx` que carga `page` desde `backend/api/pages.php`.
  - Hero/carrousel automático (forward-only) usando `page.images` y fallback a `cover_url`.
  - Sección de categorías visualmente refinada (`CategoriesSection.jsx`) con imágenes circulares, sombras, tipografías y comportamiento responsivo.
  - Estilos globales consolidados en `frontend/src/styles.css` (incluye colores, tipografías, mobile tweaks, separadores, cards, hero, etc.).
  - Formato de precios con separador de miles (`formatPrice`) y parseo robusto (`parseToIntPublic`).
  - Helpers de imagen: `getFirstImage` y `resolvePublicUrl` para normalizar rutas.
  - Cliente admin HTTP (`frontend/src/components/admin/adminApi.js`) con helpers para inventory, services y uploads.
  - Endpoint backend para subir imagen de categoría (`backend/api/upload_category_image.php`) que guarda la imagen y actualiza `pages.categories` JSON.
  - UX móvil: categorías full-bleed (100vw), scroll horizontal con scroll-snap, límite visual de 4 elementos y comportamiento wrap cuando corresponde.
  - Fuentes: Montserrat (cuerpo) y Playfair Display (títulos de categoría).
  - Separadores finos entre secciones con una clase reutilizable `.section-separator`.

  3) Diagrama de flujo (texto)
  ----------------------------

    Usuario -> [Navegador]                     
        |                                     
        | GET /:storeSlug                      
        v                                     
    [PageViewer.jsx]  -- fetch pages.php?slug --> [backend/api/pages.php]
        |                                     
        | recibe { page }                      
        v                                     
    PageViewer procesa page:                   
      - imgs = page.images || page.cover_url      
      - categoriesForSection = normalize(page.categories)
      - sections = parse(page.sections_json)     
      - si page_type contiene 'serv' -> getServices(page.id)
        |                                     
        v                                     
    Renderiza: Header, Hero (carrusel), CategoriesSection, Public Sections (productos/servicios), Footer

  Para listings (productos): [PageViewer] -> fetch inventory.php?page_id=... (+ filter params si se solicita por sección)

  import React, { useEffect, useState, useRef } from 'react'
  import * as api from './components/admin/adminApi'
  import { Link } from 'react-router-dom'
  import StoreHeader from './components/public/StoreHeader'
  import CategoriesSection from './components/CategoriesSection'

  export default function PageViewer({ slug = null }) {
    // PageViewer ahora solo maneja la home pública de la tienda: /:storeSlug
    // `slug` viene desde el wrapper de rutas en App.jsx
    const [page, setPage] = useState(null)
    const [imgIndex, setImgIndex] = useState(0)
    const [sections, setSections] = useState([])
    const [sectionProducts, setSectionProducts] = useState({})
    const [sectionLoading, setSectionLoading] = useState({})
    const [sectionError, setSectionError] = useState({})
    const [services, setServices] = useState([])
    const [searchTerm, setSearchTerm] = useState('')

    useEffect(() => {
      async function load() {
        try {
          if (slug) {
            const res = await fetch('http://localhost/LinkTiendas/Link%20Tienda/backend/api/pages.php?slug=' + encodeURIComponent(slug))
            if (!res.ok) { setPage(null); return }
            const data = await res.json()
            setPage(data.page || null)
          } else {
            const res = await fetch('http://localhost/LinkTiendas/Link%20Tienda/backend/api/pages.php')
            if (!res.ok) return
            const data = await res.json()
            setPage((data.pages && data.pages[0]) || null)
          }
        } catch (e) { }
      }
      load()
    }, [slug])

    useEffect(() => {
      if (!page) return
      let secs = []
      if (page.sections_json) {
        if (typeof page.sections_json === 'string') {
          try { secs = JSON.parse(page.sections_json) } catch (e) { secs = [] }
        } else if (Array.isArray(page.sections_json)) secs = page.sections_json
      }
      setSections(secs || [])

      if (page.page_type && page.page_type.toLowerCase().includes('serv')) {
        (async () => {
          try {
            const data = await api.getServices(page.id)
            if (data && data.ok) setServices(data.services || [])
          } catch (e) { }
        })()
      }
    }, [page])

    function resolvePublicUrl(u){
      if (!u) return null
      if (/^https?:\/\//.test(u)) return u
      const base = 'http://localhost/LinkTiendas/Link%20Tienda'
      if (u.startsWith('/')) return base + u
      return base + '/' + u
    }

    function parseToIntPublic(v){
      if (v === undefined || v === null) return 0
      if (typeof v === 'number') return Math.round(v)
      const s = String(v).trim()
      if (!s) return 0
      const lastDot = s.lastIndexOf('.')
      const lastComma = s.lastIndexOf(',')
      const decPos = Math.max(lastDot, lastComma)
      if (decPos !== -1 && s.length - decPos - 1 <= 2) {
        let cleaned = ''
        for (let i = 0; i < s.length; i++) {
          const ch = s[i]
          if (i === decPos) cleaned += '.'
          else if (ch >= '0' && ch <= '9') cleaned += ch
        }
        const num = parseFloat(cleaned)
        return isNaN(num) ? 0 : Math.round(num)
      }
      const digits = s.replace(/\D/g,'')
      return digits ? parseInt(digits,10) : 0
    }

    function formatPrice(val){
      const n = parseToIntPublic(val)
      return '$' + String(n).replace(/\B(?=(\d{3})+(?!\d))/g, '.')
    }
